import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int number = sc.nextInt();

        if (number % 10 == 7) {
           System.out.println("Lucky Ending");
        } else {
           System.out.println("Not a Lucky Ending");
        }
    }
}
