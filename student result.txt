import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        System.out.print("Enter three subject marks: ");
        int a = sc.nextInt();
        int b = sc.nextInt();
        int c = sc.nextInt();

        String result = (a >= 35 && b >= 35 && c >= 35)
                        ? "Pass" : "Fail";

        System.out.println(result);
    }
}
