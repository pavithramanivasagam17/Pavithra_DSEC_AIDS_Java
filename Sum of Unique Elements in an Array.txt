public class Main {
    public static void main(String[] args) {
        int[] a = {1, 1, 5, 3, 7, 5};
        int sum = 0;

        for (int i = 0; i < a.length; i++) {
            int count = 0;

            for (int j = 0; j < a.length; j++) {
                if (a[i] == a[j]) {
                    count++;
                }
            }

            if (count == 1) {
                sum = sum + a[i];
            }
        }

        System.out.println("Sum = " + sum);
    }
}
Sum = 10
