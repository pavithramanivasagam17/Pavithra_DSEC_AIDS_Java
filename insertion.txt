public class Main {
    public static void main(String[] args) {
        int[] a = new int[6];
        a[0] = 10;
        a[1] = 20;
        a[2] = 30;
        a[3] = 40;
        a[4] = 50;

        int n = 5;
        int pos = 2;
        int value = 25;

        for (int i = n; i > pos; i--) {
            a[i] = a[i - 1];
        }

        a[pos] = value;
        n++;

        for (int i = 0; i < n; i++) {
            System.out.print(a[i] + " ");
        }
    }
}
